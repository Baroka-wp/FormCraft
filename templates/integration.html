🚀 Endpoints API

GET Récupérer toutes les entrées
▼
{{ base_url }}/api/forms/{{ form_info.name }}/entries

📱 cURL

bash 
curl -X GET "{{ base_url }}/api/forms/{{ form_info.name }}/entries" \
  -H "Accept: application/json"

🐍 Python

python 
import requests

response = requests.get("{{ base_url }}/api/forms/{{ form_info.name }}/entries")
data = response.json()

if data['success']:
    print(f"📊 {data['count']} entrées trouvées")
    for entry in data['data']:
        print(f"- ID {entry['id']}: {entry}")
else:
    print(f"❌ Erreur: {data['error']}")

🌐 JavaScript

javascript 
fetch('{{ base_url }}/api/forms/{{ form_info.name }}/entries')
  .then(response => response.json())
  .then(data => {
    if (data.success) {
      console.log(`📊 ${data.count} entrées trouvées`);
      data.data.forEach(entry => {
        console.log(`- ID ${entry.id}:`, entry);
      });
    } else {
      console.error('❌ Erreur:', data.error);
    }
  })
  .catch(error => console.error('❌ Erreur:', error));
POST Créer une nouvelle entrée
▼
{{ base_url }}/api/forms/{{ form_info.name }}/entries

📱 cURL

bash 
curl -X POST "{{ base_url }}/api/forms/{{ form_info.name }}/entries" \
  -H "Content-Type: application/json" \
  -d '{{ example_data | tojson }}'

🐍 Python

python 
import requests
import json

data = {{ example_data | tojson(indent=4) }}

response = requests.post(
    "{{ base_url }}/api/forms/{{ form_info.name }}/entries",
    headers={"Content-Type": "application/json"},
    data=json.dumps(data)
)

result = response.json()
if result['success']:
    print("✅ Entrée créée avec succès!")
else:
    print(f"❌ Erreur: {result['error']}")
    if 'details' in result:
        for detail in result['details']:
            print(f"  - {detail}")

🌐 JavaScript

javascript 
const data = {{ example_data | tojson(indent=2) }};

fetch('{{ base_url }}/api/forms/{{ form_info.name }}/entries', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json'
  },
  body: JSON.stringify(data)
})
.then(response => response.json())
.then(result => {
  if (result.success) {
    console.log('✅ Entrée créée avec succès!');
  } else {
    console.error('❌ Erreur:', result.error);
    if (result.details) {
      result.details.forEach(detail => {
        console.error('  -', detail);
      });
    }
  }
})
.catch(error => console.error('❌ Erreur:', error));
PUT Modifier une entrée existante
▼
{{ base_url }}/api/forms/{{ form_info.name }}/entries/{id}

📱 cURL

bash 
curl -X PUT "{{ base_url }}/api/forms/{{ form_info.name }}/entries/1" \
  -H "Content-Type: application/json" \
  -d '{{ example_data | tojson }}'

🐍 Python

python 
import requests
import json

entry_id = 1  # ID de l'entrée à modifier
data = {{ example_data | tojson(indent=4) }}

response = requests.put(
    f"{{ base_url }}/api/forms/{{ form_info.name }}/entries/{entry_id}",
    headers={"Content-Type": "application/json"},
    data=json.dumps(data)
)

result = response.json()
if result['success']:
    print("✅ Entrée modifiée avec succès!")
else:
    print(f"❌ Erreur: {result['error']}")
DELETE Supprimer une entrée
▼
{{ base_url }}/api/forms/{{ form_info.name }}/entries/{id}

📱 cURL

python 
import requests

entry_id = 1  # ID de l'entrée à supprimer

if result['success']:
    print("✅ Entrée supprimée avec succès!")
else:
